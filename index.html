import requests from bs4 import BeautifulSoup from urllib.parse import urljoin import json import time MAX_GAMES = 1000 HEADERS = { "User-Agent": "Mozilla/5.0" } games = [] seen = set() def add_game(name, url, source, category="Fun"): if not name or not url: return url = url.split("#")[0] if url in seen: return if not url.startswith("http"): return seen.add(url) games.append({ "name": name.strip(), "url": url, "source": source, "category": category }) def get_page(url): try: r = requests.get(url, headers=HEADERS, timeout=15) if r.status_code == 200: return r.text except: pass return "" # -------------------------------------------------- # ADDICTING GAMES # -------------------------------------------------- print("Getting Addicting Games...") url = a.get_text(" ", strip=True) if "/game/" in href and name: add_game( name, href, "Addicting Games", "Fun" ) if len(games) >= MAX_GAMES: break # -------------------------------------------------- #
